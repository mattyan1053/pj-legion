# 話のファイルのひな形（docs/story.md の「話のファイル」）。どの物語にも並ばない見本。写して使う。
#
# - ファイルの名前（.txt を除く）が話の id。クリアの記録に書くので、遊ぶ人に配ったあとは変えない。
# - 話を足すのは、game/story/episodes/ にこのファイルを写して直し、物語のファイル（game/story/campaigns/）の [episodes] に id を 1 行足すだけ。
# - # で始まる行と空の行は読み飛ばす。[名前] の行で節を始める。節の中は「キー = 値」か、会話の行。
# - 駒・持ち物・軍・勝ち方は、名前を小文字の英語で書く（UnitKind.Id、Equipment.Id、UnitKind.Army、GameWonEvent.Reason の名前）。
#   駒: shield_trooper, sniper, commando, medic, paladin, sorcerer, spellblade, priest
#   持ち物: iron_shield, speed_boots
# - 書き間違いは、起動したときと検査（tests/game/test_episode_data.gd）が「ファイル:行: 何がおかしいか」で知らせる。
#   書き間違いのある話は、話の一覧に並ばない。

[episode]
# 会話の見出しと、話の一覧のカードに出す名前。
title = 見本の話
# 盤（手で作ったマップの id。docs/hand-map.md）。景色はマップが決める。look は無くてよい（この話だけ別の見た目にするとき。stage_look.gd の見た目の名前）。
map = ridge_path
look = stone_walls
# 無くてよい: 対戦の曲と会話の曲（sound_catalog.gd の MUSIC の名前）。無ければ、盤の見た目の曲と、会話の場面の曲。
# music = battle_tense
# talk_music = talk
# 無くてよい: 出撃の準備に出す「この話で覚えること」（チュートリアルの話だけ。48 字まで）。
# lesson = 駒を押して動かし、隣の敵を攻撃する。
# 無くてよい: 人がこのターンの終わりまで守り切れば勝ち。
# hold_turns = 10
# 無くてよい: 人の勝ち方（「,」で区切る。書けば、いつもの勝ち方を置き換える）。
#   king_defeated（将の撃破）, hq_captured（本陣の占拠）, turns_held（守り切る。hold_turns と組で書く）, routed（全滅させる）,
#   boss_slain（どちらかの中ボスを倒す）
# wins = routed

# 無くてよい節: クリアで増えるもの（docs/story.md の「解放」）。unit と item は何行でも。
[unlock]
# 新しく雇える駒（物語の軍の駒だけ）。
unit = commando
# 新しく持たせられる持ち物。
# item = speed_boots
# 部隊のコストの上限が上がる量。
budget = 2

# 人の軍。
[ally]
# 会話の話し手の名前（名前の札は青）。
name = 隊長
# 無くてよい: 話し手の姿（会話の場面に立たせる駒の形と、対戦の途中の台詞のアイコン）。無ければ将（king）。
#   駒の種類（king も魔物も書ける）。後ろに evolved を付けると進化後の形（進化する駒だけ）。
#   最後に crown を付けると、会話の姿に将の印（頭の上の金の輪）を付ける（将の姿でない将。paladin evolved crown）。
# figure = sniper evolved
# 無くてよい: ほかの話し手（何行でも）。「名前, 姿」（姿は無くてよく、無ければ将）。部隊の中の人や、教官・伝令など。
# speaker = 新兵, medic
# 無くてよい: 人の部隊を決めておく（チュートリアル。書けば出撃の準備で選び直さない）。1 行に 1 体。将は書かない。
# unit = shield_trooper
# unit = sniper + iron_shield
# 「,」のあとに、始めのレベル（1 か 2）と経験値（0 から 119）も書ける。あと少しで進化する駒から始めるとき（docs/story.md の「始めのレベル」）。
# unit = sniper, level 2, exp 110
# 無くてよい: 必殺技のゲージを、この量から始める（1 から 1000）。
# start_gauge = 800
# 無くてよい: 初期位置（将でない駒を置くマス。「x,y」を空白で区切る。何行でも。「4-8,20」で x の範囲）。無ければ、いつもの「本陣の前」。
# 書いた順に置く。部隊の数より多く書けば、出撃準備で選べる「置けるマス」が広がる（2 か所に分けてもよい）。
# deploy = 8,20  7,20  9,20  6,19  10,19  8,18

# 敵の軍。
[enemy]
# 会話の話し手の名前（名前の札は赤）。
name = 敵将
# 無くてよい: 話し手の姿（書き方は [ally] と同じ）。
figure = paladin evolved
# 無くてよい: ほかの話し手（[ally] と同じ書き方）。
# speaker = 副官, sorcerer
# 無くてよい: 敵の将の姿（見た目だけ。ルールの上では将のまま）。いま選べるのは skeleton_mage（骸骨の将。魔物の群れを率いる）。
# king_figure = skeleton_mage
# 無くてよい: none と書くと、敵は将のいない群れ（本陣は空のまま）。人の勝ち方（wins）を routed, boss_slain, turns_held から書く。
#   king_figure は書けない。敵の話し手には、将でない姿（figure）を書く。
# king = none
# 敵の軍（military, fantasy, none）。必殺技はこの軍のもの。none は軍に属さない敵（魔物の群れ。必殺技もゲージも無い）。
army = fantasy
# 敵の部隊。1 行に 1 体、並びの順に初期位置へ置く。持ち物は「+」でつなぐ（魔物は持てない）。将は書かない（いつも本陣に置く）。
# 魔物（skeleton, skeleton_archer, skeleton_brute）も書ける（docs/unit-roster.md の「魔物」）。
unit = paladin + iron_shield
unit = sorcerer
unit = priest
# 無くてよい: 初期位置（[ally] と同じ書き方。部隊の数以上）。
# deploy = 8,9  5,9  11,9

# 対戦の前の会話。1 行が 1 つの文で「話し手: 文」（話し手は [ally] か [enemy] の name か speaker）。10 行まで、1 行は 60 字まで。
# 人の軍の話し手は左、敵の軍の話し手は右に立つ（同じ側に何人かいれば、話した人に入れ替わる）。
# 敵の軍が出てこない会話は、最初に話す人が左、ほかの人が右。
[before]
隊長: 見本の話だ。ここに会話を書く。
敵将: かかってこい。

# 対戦に勝ったあとの会話（書き方は [before] と同じ）。解放の知らせは、この後ろに自動で出る。
[after]
隊長: 見本の話は、ここまで。

# 無くてよい節: 対戦の途中の台詞（対戦を止めずに数秒出る）。「(きっかけ 値) 話し手: 文」。3 つまで、文は 40 字まで。
#   (turn N)          ターン N の、人の番の始め（turn 1 は対戦の始め）
#   (king_half 陣営)   その陣営の将の HP が半分を切った（陣営は ally か enemy）
#   (fallen 陣営)      その陣営の駒（将を除く）が初めて倒れた
#   (ult_ready 陣営)   その陣営の必殺技のゲージがたまった
#   (relic 陣営)       その陣営が聖遺物を受け取った
[cues]
(turn 1) 隊長: 前へ出すぎるな。
(king_half enemy) 敵将: まだだ。
