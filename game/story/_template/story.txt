# 物語のファイルの見本（docs/story.md の「物語を 1 つ足す」）。このフォルダを写して使う。
#
# 物語を足すには:
#   1. このフォルダ（_template）を、game/story/ の下に別の名前で写す（英小文字と _。例: game/story/beast_tribe/）。
#      フォルダの名前が物語の id になる（クリアの記録に書くので、配ったあとは変えない）。
#      名前が _ で始まるフォルダはゲームが読まない（この見本が一覧に出ないのは、そのため）。
#   2. この story.txt を直す。
#   3. 話のファイル（<話の id>.txt）を同じフォルダに置き、下の [episodes] に話の id を遊ぶ順に並べる。
#   4. それだけで、物語を選ぶ場面に並ぶ（どこにも登録しない）。
#
# 書き方: # で始まる行と空の行は読み飛ばす。[名前] の行で節を始める。節の中は「キー = 値」。
# 書き間違いは、起動したときと検査（tools/check.sh）が「ファイル:行: 何がおかしいか」で知らせる。

[campaign]
# 物語を選ぶ場面と、話の一覧の見出しに出す名前。
title = 見本の物語
# 遊ぶ人が使う軍（military か fantasy）。クリアで増やせるのは、この軍の駒だけ。
army = military
# この物語の部隊の決まりの名前（取っておく部隊に書く）。ほかの物語と重ねない。物語の id と同じでよい。配ったあとは変えない。
rules = template
# 最初のコスト上限（部隊の駒と持ち物のコストの合計の上限）。
budget = 20
# 無くてよい: 物語を選ぶ場面での並びの順（小さいほうが先）。書かなければ、順を書いた物語の後ろに、id の順で並ぶ。
order = 9
# 物語の紹介の 1 行（40 字まで。話のある物語には書く。検査が見る）。物語を選ぶ場面のカードに出す。遊ぶ人が誰として、何をする物語か。
intro = 新任の少尉として、見本の尾根を越える。

# 最初から雇える駒（この物語の軍の駒。何行でも）と、持たせられる持ち物（無くてよい）。
#   MILITARY の駒: shield_trooper, sniper, commando, medic
#   FANTASY の駒:  paladin, sorcerer, spellblade, priest
#   持ち物: iron_shield, speed_boots など（一覧は docs/player-loadout.md の「装備」）
[start]
unit = shield_trooper
unit = sniper
# item = iron_shield

# 主な人物の節（話の一覧の頭に、姿の額・名前・立ち位置で出す。話のある物語には書く。検査が見る）。1 行に 1 人で「名前 = 姿, 立ち位置」。6 人まで。
#   姿は話のファイルの [cast] と同じ書き方（king、sniper、paladin evolved crown など）。立ち位置は 10 字まで。
#   遊ぶ人の将と、隊の人を書く（敵は書かない）。名前と姿は、話の [cast] の ally の話し手と同じにする（検査が見る）。
[cast]
少尉 = king, 新任の少尉
新兵 = sniper, 新兵の狙撃手

# 話の並び（一本道）。1 行に 1 つ、話のファイルの名前（.txt を除く）。上から順に遊ぶ。
[episodes]
first_battle
scenes
